```python
# algorithms/equivalence_partitioning.py
def partition_text_input(element: Element) -> List[Partition]:
    """Partition text input into equivalence classes"""
    partitions = []
    
    # Valid partition (within constraints)
    if element.valid_values:
        partitions.append(Partition(
            name="valid",
            representative=element.valid_values[0],
            expected_behavior="accepted"
        ))
    
    # Boundary partition (at limits)
    if element.constraints:
        partitions.append(Partition(
            name="boundary",
            representative="a" * element.constraints.max_length,
            expected_behavior="accepted_at_limit"
        ))
    
    # Invalid partition
    if element.invalid_values:
        partitions.append(Partition(
            name="invalid",
            representative=element.invalid_values[0],
            expected_behavior="rejected"
        ))
    
    return partitions

# algorithms/boundary_value_analysis.py
def get_boundary_values(constraint: Constraint) -> List[Tuple[Any, str]]:
    """Get boundary test values"""
    values = []
    
    if constraint.min_value is not None:
        values.append((constraint.min_value - 1, "below_min"))
        values.append((constraint.min_value, "at_min"))
        values.append((constraint.min_value + 1, "above_min"))
    
    if constraint.max_value is not None:
        values.append((constraint.max_value - 1, "below_max"))
        values.append((constraint.max_value, "at_max"))
        values.append((constraint.max_value + 1, "above_max"))
    
    return values
```

```python
class CoverageCalculator:
    def __init__(self, phase1_parser: Phase1Parser, tests: List[TestCase]):
        """Initialize with Phase 1 data and test cases"""
        
    def calculate_coverage(self) -> CoverageMetrics:
        """Calculate all coverage metrics"""
        element_cov = self.calculate_element_coverage()
        path_cov = self.calculate_path_coverage()
        scenario_cov = self.calculate_scenario_coverage()
        blended = self.blend_coverage(element_cov, path_cov, scenario_cov)
        
        return CoverageMetrics(
            element_coverage=element_cov,
            path_coverage=path_cov,
            scenario_coverage=scenario_cov,
            blended_coverage=blended
        )
        
    def calculate_element_coverage(self) -> float:
        """(tested_elements / total_elements) * 100"""
        tested_ids = set()
        for test in self.tests:
            tested_ids.update(test.elements_used)
        
        total_elements = len(self.phase1_parser.get_all_elements())
        return (len(tested_ids) / total_elements) * 100
        
    def calculate_path_coverage(self) -> float:
        """(tested_workflows / total_workflows) * 100"""
        tested_workflows = set()
        for test in self.tests:
            if test.algorithm_used == "state":
                tested_workflows.add(test.title)
        
        total_workflows = len(self.phase1_parser.get_workflows())
        return (len(tested_workflows) / total_workflows) * 100
        
    def calculate_scenario_coverage(self) -> float:
        """(tested_scenarios / reachable_scenarios) * 100"""
        tested_scenarios = 0
        for test in self.tests:
            # Each test covers coverage_contribution points
            tested_scenarios += test.coverage_contribution
        
        # Total scenarios = sum of all possible partitions + boundaries
        total_scenarios = self._calculate_total_scenarios()
        return (tested_scenarios / total_scenarios) * 100
        
    def blend_coverage(self, element: float, path: float, 
                      scenario: float) -> float:
        """Blended = 0.4×element + 0.35×path + 0.25×scenario"""
        return (0.4 * element) + (0.35 * path) + (0.25 * scenario)
        
    def get_coverage_by_element_type(self) -> Dict[str, float]:
        """Coverage breakdown by element type"""
        # Group elements by type
        # Calculate coverage for each group
        
    def identify_coverage_gaps(self) -> List[CoverageGap]:
        """Identify elements not covered"""
        covered_ids = set()
        for test in self.tests:
            covered_ids.update(test.elements_used)
        
        all_elements = self.phase1_parser.get_all_elements()
        gaps = []
        for elem in all_elements:
            if elem.element_id not in covered_ids:
                gaps.append(CoverageGap(
                    element_id=elem.element_id,
                    element_label=elem.label,
                    confidence=elem.locator_confidence,
                    reason="Not covered"
                ))
        
        return gaps
```

```python
class CoverageVisualizer:
    def __init__(self, coverage_metrics: CoverageMetrics, 
                 phase1_parser: Phase1Parser):
        """Initialize with coverage data"""
        
    def generate_heatmap_data(self) -> Dict:
        """Generate JSON data for heatmap visualization"""
        return {
            "elements": [
                {
                    "element_id": elem.element_id,
                    "label": elem.label,
                    "type": elem.type,
                    "coverage": self._get_element_coverage(elem.element_id),
                    "color": self._coverage_to_color(coverage),
                    "test_count": self._count_tests_for_element(elem.element_id),
                    "importance": elem.importance_rank
                }
                for elem in phase1_parser.get_all_elements()
            ],
            "summary": {
                "blended_coverage": coverage_metrics.blended_coverage,
                "element_coverage": coverage_metrics.element_coverage,
                "path_coverage": coverage_metrics.path_coverage,
                "scenario_coverage": coverage_metrics.scenario_coverage
            }
        }
        
    def _coverage_to_color(self, coverage: float) -> str:
        """Map coverage percentage to color"""
        if coverage >= 90:
            return "#00AA00"  # Green
        elif coverage >= 75:
            return "#FFAA00"  # Orange
        else:
            return "#FF0000"  # Red
            
    def generate_element_heatmap(self) -> Dict:
        """Heatmap showing coverage by element type"""
        return {
            "by_type": {
                element_type: {
                    "coverage": self._get_type_coverage(element_type),
                    "test_count": count
                }
                for element_type, count in self._group_by_type().items()
            }
        }
```

```python
class HITLManager:
    def __init__(self, test_generator: TestGenerator):
        """Initialize with test generator"""
        
    def run_hitl_loop(self, rounds: int = 2) -> HITLResult:
        """Run complete HITL feedback loop"""
        feedback_rounds = []
        
        for round_num in range(1, rounds + 1):
            tests = self.test_generator.generate_tests()
            feedback = self.present_and_collect_feedback(tests, round_num)
            feedback_rounds.append(feedback)
            
            # Learn from feedback
            patterns = self.extract_patterns(feedback)
            self.apply_learning(patterns)
        
        return HITLResult(feedback_rounds=feedback_rounds)
        
    def present_and_collect_feedback(self, tests: List[TestCase], 
                                     round_num: int) -> Dict:
        """Present tests to human and collect feedback"""
        # For each test:
        #   - Print test details
        #   - Get human verdict (approve/reject/modify)
        #   - Record feedback
        
        feedback_log = []
        for test in tests:
            print(f"\n--- Test {test.test_id} ---")
            print(f"Title: {test.title}")
            print(f"Algorithm: {test.algorithm_used}")
            print(f"Coverage: {test.coverage_contribution}%")
            print(f"Steps: {len(test.test_steps)}")
            
            verdict = input("Verdict (approve/reject/modify)? ").lower()
            comment = input("Comment? ")
            
            feedback_log.append({
                "test_id": test.test_id,
                "verdict": verdict,
                "comment": comment,
                "element_id": test.elements_used[0],
                "algorithm": test.algorithm_used
            })
        
        return feedback_log
        
    def extract_patterns(self, feedback: List[Dict]) -> List[str]:
        """Extract patterns from feedback"""
        patterns = []
        
        # Pattern 1: Which algorithms have highest approval?
        algo_approval = {}
        for f in feedback:
            algo = f["algorithm"]
            verdict = f["verdict"]
            if algo not in algo_approval:
                algo_approval[algo] = {"approve": 0, "reject": 0}
            
            if verdict == "approve":
                algo_approval[algo]["approve"] += 1
            else:
                algo_approval[algo]["reject"] += 1
        
        for algo, counts in algo_approval.items():
            approval_rate = counts["approve"] / (counts["approve"] + counts["reject"])
            if approval_rate > 0.90:
                patterns.append(f"{algo}_high_approval_{approval_rate:.0%}")
            elif approval_rate < 0.70:
                patterns.append(f"{algo}_low_approval_{approval_rate:.0%}")
        
        return patterns
        
    def apply_learning(self, patterns: List[str]):
        """Apply learning to next round"""
        # For each pattern:
        #   - Increase/decrease frequency of algorithm
        #   - Adjust test complexity
        #   - Modify selection criteria
        
        for pattern in patterns:
            if "high_approval" in pattern:
                algo = pattern.split("_")[0]
                self.test_generator.increase_algorithm_frequency(algo)
            elif "low_approval" in pattern:
                algo = pattern.split("_")[0]
                self.test_generator.decrease_algorithm_frequency(algo)
```

```python
class HallucinationDetector:
    def __init__(self, phase1_parser: Phase1Parser):
        """Initialize with Phase 1 data"""
        
    def validate_test(self, test: TestCase) -> bool:
        """Validate test for hallucinations"""
        # Check 1: All element_ids exist in Phase 1
        # Check 2: All locators match Phase 1
        # Check 3: Confidence scores are valid
        # Check 4: Test steps are realistic
        
        for step in test.test_steps:
            if not self._element_exists(step.element_id):
                raise HallucinationError(f"Element {step.element_id} not found in Phase 1")
            
            if not self._locator_valid(step.element_id, step.locator):
                raise HallucinationError(f"Locator {step.locator} not valid for {step.element_id}")
        
        return True
        
    def _element_exists(self, element_id: str) -> bool:
        """Check if element exists in Phase 1"""
        try:
            self.phase1_parser.get_element_by_id(element_id)
            return True
        except:
            return False
```

```python
    hitl = HITLManager(generator)
    hitl_result = hitl.run_hitl_loop(rounds=2)
    print(f"✅ Completed HITL 2.0 loop")
    
    # Step 6: Save outputs
    save_test_cases(tests, "test_cases.json")
    save_coverage(coverage, "coverage_metrics.json")
    save_heatmap(heatmap, "coverage_heatmap.json")
    print(f"✅ All outputs saved")
    
    return tests, coverage, hitl_result
```
